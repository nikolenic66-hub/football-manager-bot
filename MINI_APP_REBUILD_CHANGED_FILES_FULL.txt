def retry_on_interface_error(endpoint):
    """Retry an HTTP endpoint once if its DB connection was closed unexpectedly."""
    @wraps(endpoint)
    async def wrapped(*args, **kwargs):
        try:
            return await endpoint(*args, **kwargs)
        except InterfaceError:
            log.warning("DB connection closed, retrying...")
            return await endpoint(*args, **kwargs)
    return wrapped


async def authenticated_telegram_user(x_telegram_init_data: str | None = Header(default=None)) -> dict[str, Any]:
    """Validate signed Telegram initData; never accept unsigned user IDs."""
    init_data = x_telegram_init_data.strip() if isinstance(x_telegram_init_data, str) else ""
    try:
        return validate_telegram_init_data(init_data, settings.bot_token)
    except TelegramInitDataError as exc:
        # Log only a reason code. Never log the raw initData, hash, token, or user payload.
        reason = str(exc)
        log.warning("Telegram Mini App authentication rejected (reason=%s)", reason)
        if reason == "missing":
            detail = "Откройте приложение из Telegram: initData отсутствует."
        elif reason == "bot_token_missing":
            detail = "На сервере не настроен BOT_TOKEN."
        elif reason == "signature_mismatch":
            detail = "Invalid Telegram initData signature. Проверьте, что BOT_TOKEN в Amvera совпадает с токеном этого бота."
        elif reason == "expired":
            detail = "Telegram initData устарел. Полностью закройте и снова откройте Mini App из Telegram."
        else:
            detail = "Invalid Telegram initData (reason: %s). Откройте Mini App заново из Telegram." % reason
        raise HTTPException(status_code=401, detail=detail) from exc

async def next_match(user=Depends(authenticated_telegram_user)):
    """Return the next scheduled or currently live match for the authenticated user's club."""
    async with SessionLocal() as session:
        club = await user_and_club(user, session)
        row = (await session.execute(text("""SELECT m.id, m.home_club_id, m.away_club_id,
                hc.name AS home_name, ac.name AS away_name,
                COALESCE(m.home_score, 0) AS home_score, COALESCE(m.away_score, 0) AS away_score,
                m.status, m.current_minute, m.started_at, m.scheduled_at,
                m.competition_type, m.is_test,
                CASE WHEN m.competition_type='FRIENDLY' THEN 'Товарищеский матч'
                     ELSE COALESCE(l.name, m.competition_type, 'Матч') END AS competition_name
            FROM matches m
            JOIN clubs hc ON hc.id = m.home_club_id
            JOIN clubs ac ON ac.id = m.away_club_id
            LEFT JOIN leagues l ON l.id = m.league_id
            WHERE (m.home_club_id = :club_id OR m.away_club_id = :club_id)
              AND (m.status='LIVE' OR (m.status='SCHEDULED' AND (m.scheduled_at IS NULL OR m.scheduled_at >= now())))
            ORDER BY CASE WHEN m.status='LIVE' THEN 0 ELSE 1 END,
                     m.scheduled_at ASC NULLS LAST,
                     m.id ASC
            LIMIT 1"""), {"club_id": int(club["id"])})).mappings().first()
        if row is None:
            return {"match": None}
        match = dict(row)
        if match["status"] == "LIVE" or (match.get("is_test") and match.get("competition_type") == "FRIENDLY"):
            events = (await session.execute(text("""SELECT id, minute, event_type, club_id,
                    player_id, description, metadata
                FROM match_events WHERE match_id = :match_id ORDER BY minute, id"""),
                {"match_id": int(match["id"])})).mappings().all()
            match["events"] = [dict(event) for event in events]
        else:
            match["events"] = []
        return {"match": match}

        club=await user_and_club(user, session)

            listing = (await session.execute(text("""SELECT id,seller_club_id,asking_price FROM transfer_listings
                WHERE player_id=:p AND status='LISTED' FOR UPDATE"""), {"p": payload.player_id})).mappings().first()

                if listing["seller_club_id"] is not None and int(listing["seller_club_id"]) == int(club["id"]):

            raise HTTPException(status_code=400, detail=str(exc)) from exc

        club = await user_and_club(user, session)
        row = (await session.execute(text("""SELECT p.id,p.first_name,p.last_name,p.position,p.age,p.nationality,
                p.market_value,p.potential,p.rarity,p.pace,p.shooting,p.passing,p.dribbling,p.defending,p.physical,p.stamina,p.mental,

            raise HTTPException(status_code=400, detail="Этот игрок уже есть в вашем клубе")

        labels = {"pace":"Скорость","shooting":"Удар","passing":"Пас","dribbling":"Дриблинг","defending":"Защита","physical":"Физика","stamina":"Выносливость","mental":"Ментальность"}

            "strengths": [{"attribute": labels[k], "value": v} for k,v in strengths],
            "weaknesses": [{"attribute": labels[k], "value": v} for k,v in weaknesses],
            "potential_gap": max(0, int(data["potential"] or 0)-int(data["overall"] or 0)),
            "recommendation": "Высокий приоритет" if int(data["potential"] or 0)-int(data["overall"] or 0) >= 15 else "Рассмотреть по потребностям состава"}

@app.websocket("/ws")
async def match_events_socket(websocket: WebSocket):
    # WebSocket authentication uses the same signature and freshness checks as HTTP.
    init_data = websocket.query_params.get("initData", "")
    try:
        verified_user = validate_telegram_init_data(init_data, settings.bot_token)
    except TelegramInitDataError as exc:
        log.warning("WebSocket Telegram authentication rejected (reason=%s)", str(exc))
        await websocket.close(code=4401)
        return

    await websocket.accept()
    try:
        while True:
            try:
                async with SessionLocal() as session:
                    club=await user_and_club({"telegram_id":verified_user["telegram_id"]}, session)
                    rows=(await session.execute(text("""SELECT m.id,m.home_club_id,m.away_club_id,m.home_score,m.away_score,
                        m.status FROM matches m WHERE m.status='LIVE' AND (m.home_club_id=:c OR m.away_club_id=:c)"""), {"c":club["id"]})).mappings().all()
                    output=[]
                    for match in rows:
                        events=(await session.execute(text("SELECT id,minute,event_type,club_id,player_id,description,metadata FROM match_events WHERE match_id=:m ORDER BY id DESC LIMIT 20"), {"m":match["id"]})).mappings().all()
                        output.append({"match":dict(match),"events":[dict(e) for e in reversed(events)]})
                    await websocket.send_text(json.dumps({"matches":output}, default=str))
            except InterfaceError:
                log.warning("DB connection closed, retrying...")
                await asyncio.sleep(0.1)
                continue
            await asyncio.sleep(2)
    except WebSocketDisconnect:
        return
    except Exception:
        log.exception("WebSocket match feed failed")
        try: await websocket.close(code=1011)
        except Exception: pass

MSK = timezone(timedelta(hours=3))


class SeasonState(str, Enum):
    PRESEASON = 'PRESEASON'
    ACTIVE = 'ACTIVE'
    SEASON_END = 'SEASON_END'
    OFFSEASON = 'OFFSEASON'
    CONTRACT_WINDOW = 'CONTRACT_WINDOW'


_STATE_FLOW = {
    SeasonState.PRESEASON: SeasonState.ACTIVE,
    SeasonState.ACTIVE: SeasonState.SEASON_END,
    SeasonState.SEASON_END: SeasonState.OFFSEASON,
    SeasonState.OFFSEASON: SeasonState.CONTRACT_WINDOW,
    SeasonState.CONTRACT_WINDOW: SeasonState.PRESEASON,
}


def next_season_state(state: SeasonState | str) -> SeasonState:
    current = state if isinstance(state, SeasonState) else SeasonState(str(state).upper())
    return _STATE_FLOW[current]


def season_finished(match_statuses: Iterable[str]) -> bool:
    statuses = list(match_statuses)
    return bool(statuses) and all(str(status).upper() in {'FINISHED', 'FAILED'} for status in statuses)


def deterministic_match_seed(season_seed: int, match_id: int) -> int:
    return int(season_seed) + int(match_id)


def new_season_seed() -> int:
    return randbelow(1_000_000_000)

        SELECT id,sequence_id,season_number,state,season_seed,started_at,ended_at,offseason_at,preseason_at,champion_club_id
        FROM seasons ORDER BY id DESC LIMIT 1 FOR UPDATE
    """))).mappings().first()


async def check_season_complete(session) -> bool:
    season = await _current_season(session)
    if not season:
        return False
    if season['state'] == SeasonState.PRESEASON.value:
        await session.execute(text("UPDATE seasons SET state='ACTIVE',started_at=COALESCE(started_at,CAST(:now AS timestamptz)) WHERE id=:s"),
                              {'s': int(season['id']), 'now': datetime.now(timezone.utc)})
        return False
    if season['state'] == SeasonState.SEASON_END.value:
        return await run_season_end(session, int(season['id']))
    if season['state'] == SeasonState.OFFSEASON.value:
        await session.execute(text("UPDATE seasons SET state='CONTRACT_WINDOW' WHERE id=:s"), {'s':int(season['id'])})
        return True
    if season['state'] == SeasonState.CONTRACT_WINDOW.value:
        return await advance_contract_window(session)
    if season['state'] != SeasonState.ACTIVE.value:
        return False
    total = int((await session.execute(text("SELECT COUNT(*) FROM matches WHERE season_id=:s"), {'s': int(season['id'])})).scalar_one())
    pending = int((await session.execute(text("SELECT COUNT(*) FROM matches WHERE season_id=:s AND status NOT IN ('FINISHED','FAILED')"), {'s': int(season['id'])})).scalar_one())
    if total == 0 or pending:
        return False
    # A season with a damaged/missing league round must not be closed merely
    # because every match row that still exists is terminal.
    active_leagues=int((await session.execute(text("""SELECT COUNT(*) FROM leagues l
        WHERE l.status='ACTIVE' AND (l.season_id=:s OR EXISTS (
            SELECT 1 FROM matches m WHERE m.league_id=l.id AND m.season_id=:s
        ))"""), {'s': int(season['id'])})).scalar_one())
    if active_leagues:
        return False
    return await run_season_end(session, int(season['id']))


async def _snapshot_league(session, league_id, season_id):
    rows = (await session.execute(text("""
        SELECT t.club_id,c.name club_name,t.played,t.wins,t.draws,t.losses,
               t.goals_for,t.goals_against,t.points
        FROM league_teams t JOIN clubs c ON c.id=t.club_id
        WHERE t.league_id=:l
        ORDER BY t.points DESC,(t.goals_for-t.goals_against) DESC,t.goals_for DESC,c.id
    """), {'l': league_id})).mappings().all()
    league = (await session.execute(text("SELECT name,season_number FROM leagues WHERE id=:l"), {'l': league_id})).mappings().first()
    if not league or not rows:
        return list(rows)
    season_meta = (await session.execute(text("SELECT sequence_id,season_number FROM seasons WHERE id=:s"), {'s': season_id})).mappings().first()
    snapshot_payload = [dict(r) for r in rows]
    await session.execute(text("""
        INSERT INTO season_league_snapshots(sequence_id,season_id,league_id,league_name,tier,season_number,standings)
        VALUES(:seq,:s,:l,:name,:tier,:n,CAST(:standings AS jsonb))
        ON CONFLICT(season_id,league_id) DO NOTHING
    """), {
        'seq': int(season_meta['sequence_id']), 's': season_id, 'l': league_id,
        'name': str(league['name']), 'tier': str(league.get('tier') or ''),
        'n': int(league['season_number'] or 1), 'standings': json.dumps(snapshot_payload, default=str, ensure_ascii=False),
    })
    await session.execute(text("""
        INSERT INTO season_history(league_id,season_id,season_number,champion_club_id,promoted_club_ids,relegated_club_ids)
        VALUES(:l,:s,:n,:champ,'[]'::jsonb,'[]'::jsonb)
        ON CONFLICT(league_id,season_number) DO UPDATE SET
            season_id=EXCLUDED.season_id, champion_club_id=EXCLUDED.champion_club_id
    """), {'l': league_id, 's': season_id, 'n': int(league['season_number'] or 1), 'champ': int(rows[0]['club_id'])})
    return list(rows)


def _promotion_pairs(leagues):
    tiers = ['ROOKIE', 'BRONZE', 'SILVER', 'GOLD', 'DIAMOND', 'ELITE', 'LEGEND']
    by_tier = {}
    for league in leagues:
        by_tier.setdefault(str(league.get('tier') or 'ROOKIE').upper(), []).append(league)
    pairs = []
    for lower_tier, upper_tier in zip(tiers, tiers[1:]):
        for lower in sorted(by_tier.get(lower_tier, []), key=lambda x: int(x['id'])):
            for upper in sorted(by_tier.get(upper_tier, []), key=lambda x: int(x['id'])):
                pairs.append((lower, upper))
    return pairs


async def _apply_promotion_relegation(session, leagues, standings, season_id):
    for lower, upper in _promotion_pairs(leagues):
        low_rows = standings.get(int(lower['id']), [])
        up_rows = standings.get(int(upper['id']), [])
        if len(low_rows) < 4 or len(up_rows) < 4:
            continue
        promoted = [int(r['club_id']) for r in low_rows[:2]]
        relegated = [int(r['club_id']) for r in up_rows[-2:]]
        for cid in promoted:
            await session.execute(text("DELETE FROM league_teams WHERE league_id=:l AND club_id=:c"), {'l': int(lower['id']), 'c': cid})
        for cid in relegated:
            await session.execute(text("DELETE FROM league_teams WHERE league_id=:l AND club_id=:c"), {'l': int(upper['id']), 'c': cid})
        for cid in promoted:
            await session.execute(text("INSERT INTO league_teams(league_id,club_id) VALUES(:l,:c) ON CONFLICT DO NOTHING"), {'l': int(upper['id']), 'c': cid})
        for cid in relegated:
            await session.execute(text("INSERT INTO league_teams(league_id,club_id) VALUES(:l,:c) ON CONFLICT DO NOTHING"), {'l': int(lower['id']), 'c': cid})
        await session.execute(text("""
            UPDATE season_history SET promoted_club_ids=CAST(:p AS jsonb),relegated_club_ids=CAST(:r AS jsonb)
            WHERE league_id=:l AND season_id=:s
        """), {'p': json.dumps(promoted), 'r': json.dumps(relegated), 'l': int(lower['id']), 's': season_id})
        await session.execute(text("""
            UPDATE season_history SET promoted_club_ids=CAST(:p AS jsonb),relegated_club_ids=CAST(:r AS jsonb)
            WHERE league_id=:l AND season_id=:s
        """), {'p': json.dumps(relegated), 'r': json.dumps(promoted), 'l': int(upper['id']), 's': season_id})


async def _apply_player_settlement(session, season_id):
    await session.execute(text("""
        INSERT INTO player_season_state(player_id,season_id,aged,development_applied,contract_processed,reward_claimed)
        SELECT cp.player_id,:s,FALSE,FALSE,FALSE,FALSE FROM club_players cp
        ON CONFLICT(player_id,season_id) DO NOTHING
    """), {'s': season_id})
    await session.execute(text("""
        UPDATE players p SET age=LEAST(40,p.age+1)
        WHERE EXISTS (SELECT 1 FROM player_season_state ps WHERE ps.player_id=p.id AND ps.season_id=:s AND ps.aged=FALSE)
    """), {'s': season_id})
    await session.execute(text("UPDATE player_season_state SET aged=TRUE,development_applied=TRUE WHERE season_id=:s AND aged=FALSE"), {'s': season_id})
    await session.execute(text("""
        UPDATE club_players cp SET
            contract_until=GREATEST(COALESCE(cp.contract_until,CURRENT_DATE),CURRENT_DATE)+INTERVAL '1 year',
            contract_salary=COALESCE(cp.contract_salary,(SELECT p.salary FROM players p WHERE p.id=cp.player_id))
        WHERE cp.contract_until IS NULL OR cp.contract_until<=CURRENT_DATE+INTERVAL '180 days'
          AND EXISTS (SELECT 1 FROM player_season_state ps WHERE ps.player_id=cp.player_id AND ps.season_id=:s AND ps.contract_processed=FALSE)
    """), {'s': season_id})
    await session.execute(text("UPDATE player_season_state SET contract_processed=TRUE WHERE season_id=:s AND contract_processed=FALSE"), {'s': season_id})


async def _settle_rewards(session, leagues, season_id):
    from app.services import settle_season_prizes, apply_season_reputation
    for league in leagues:
        lid = int(league['id'])
        await apply_season_reputation(session, lid)
        await settle_season_prizes(session, lid)
    # Prize settlement is guarded by the financial transaction marker; this flag
    # records that the player-side reward checkpoint has also been completed.
    await session.execute(text("UPDATE player_season_state SET reward_claimed=TRUE WHERE season_id=:s AND reward_claimed=FALSE"), {'s': season_id})


async def run_season_end(session, season_id: int | None = None) -> bool:
    season = (await session.execute(text("SELECT id,season_number,state FROM seasons WHERE id=COALESCE(:s,(SELECT id FROM seasons ORDER BY id DESC LIMIT 1)) FOR UPDATE"), {'s': season_id})).mappings().first()
    if not season or season['state'] not in (SeasonState.ACTIVE.value, SeasonState.SEASON_END.value):
        return False
    season_id = int(season['id'])
    if season['state'] == SeasonState.ACTIVE.value:
        await session.execute(text("UPDATE seasons SET state='SEASON_END',ended_at=COALESCE(ended_at,CAST(:now AS timestamptz)) WHERE id=:s"), {'s': season_id, 'now': datetime.now(timezone.utc)})
    leagues = (await session.execute(text("""
        SELECT id,name,tier,season_number,season_id FROM leagues
        WHERE season_id=:s AND name NOT LIKE 'FM27 Quick Test%' ORDER BY tier,id FOR UPDATE
    """), {'s': season_id})).mappings().all()
    standings = {int(l['id']): await _snapshot_league(session, int(l['id']), season_id) for l in leagues}
    await _settle_rewards(session, leagues, season_id)
    await _apply_promotion_relegation(session, leagues, standings, season_id)
    await _apply_player_settlement(session, season_id)
    champion = None
    for tier in ('LEGEND','ELITE','DIAMOND','GOLD','SILVER','BRONZE','ROOKIE'):
        top = next((standings[int(l['id'])][0] for l in leagues if str(l['tier'] or '').upper() == tier and standings.get(int(l['id']))), None)
        if top:
            champion = int(top['club_id']); break
    now = datetime.now(timezone.utc)
    await session.execute(text("UPDATE seasons SET champion_club_id=:c,state='OFFSEASON',offseason_at=COALESCE(offseason_at,CAST(:now AS timestamptz)),preseason_at=COALESCE(preseason_at,CAST(:until AS timestamptz)) WHERE id=:s"),
                          {'c': champion, 's': season_id, 'now': now, 'until': now + timedelta(days=3)})
    from app.game.event_queue import queue_event
    await queue_event(session, 'CONTRACT_WINDOW_OPEN', now, {'season_id': season_id})
    return True


async def advance_contract_window(session, force=False):
    season = (await session.execute(text("SELECT id,state,preseason_at FROM seasons ORDER BY id DESC LIMIT 1 FOR UPDATE"))).mappings().first()
    if not season or season['state'] != SeasonState.CONTRACT_WINDOW.value:
        return False
    if not force and season['preseason_at'] is not None and season['preseason_at'] > datetime.now(timezone.utc):
        return False
    await start_new_season(session, int(season['id']))
    return True




async def ensure_initial_season(session) -> int:
    """Create the first PRESEASON record when a fresh/reset database has none."""
    existing = (await session.execute(text("SELECT id FROM seasons ORDER BY id DESC LIMIT 1"))).scalar_one_or_none()
    if existing is not None:
        return int(existing)
    seed = new_season_seed()
    sequence_id = int((await session.execute(text("""
        INSERT INTO season_sequences(root_seed, current_season_id)
        VALUES(:seed, NULL) RETURNING id
    """), {"seed": seed})).scalar_one())
    season_id = int((await session.execute(text("""
        INSERT INTO seasons(sequence_id, season_number, state, season_seed, started_at)
        VALUES(:seq, 1, 'PRESEASON', :seed, CAST(:now AS timestamptz)) RETURNING id
    """), {"seq": sequence_id, "seed": seed, "now": datetime.now(timezone.utc)})).scalar_one())
    await session.execute(text("UPDATE season_sequences SET current_season_id=:s WHERE id=:seq"),
                          {"s": season_id, "seq": sequence_id})
    await session.execute(text("""
        UPDATE leagues SET season_id=:s, season_number=1, season_state='PRESEASON'
        WHERE season_id IS NULL
    """), {"s": season_id})
    return season_id


async def start_new_season(session, previous_season_id: int | None = None) -> int:
    """Create a season and all fixtures atomically within the caller transaction."""
    async with session.begin_nested():
        return await _start_new_season_in_transaction(session, previous_season_id)


async def _start_new_season_in_transaction(session, previous_season_id: int | None = None) -> int:
    previous = None
    if previous_season_id is not None:
        previous = (await session.execute(text("SELECT id,sequence_id,season_number FROM seasons WHERE id=:s"), {'s': previous_season_id})).mappings().first()
    else:
        previous = (await session.execute(text("SELECT id,sequence_id,season_number FROM seasons ORDER BY id DESC LIMIT 1"))).mappings().first()
    if not previous:
        raise ValueError('Cannot start a new season without an existing season sequence')
    previous_season_id = int(previous['id'])
    sequence_id = int(previous['sequence_id'])
    next_number = int(previous['season_number']) + 1
    now = datetime.now(MSK)
    seed = new_season_seed()
    new_id = int((await session.execute(text("""
        INSERT INTO seasons(sequence_id,season_number,state,season_seed,started_at)
        VALUES(:seq,:n,'PRESEASON',:seed,CAST(:started AS timestamptz)) RETURNING id
    """), {'seq': sequence_id, 'n': next_number, 'seed': seed, 'started': now})).scalar_one())
    old_league_ids=[int(x) for x in (await session.execute(text(
        "SELECT id FROM leagues WHERE season_id=:old OR (season_id IS NULL AND season_number=:n) ORDER BY id FOR UPDATE"
    ), {'old': previous_season_id, 'n': int(previous['season_number'])})).scalars().all()]
    # Remove old season competitions and their match links before deleting its
    # fixtures. This is required by the actual FK graph (cup/CL match links use
    # ON DELETE SET NULL; snapshots intentionally remain immutable history).
    await session.execute(text("DELETE FROM cup_matches WHERE cup_id IN (SELECT id FROM cups WHERE season_id=:old OR season=:n)"),
                          {'old': previous_season_id, 'n': int(previous['season_number'])})
    await session.execute(text("DELETE FROM cups WHERE season_id=:old OR season=:n"),
                          {'old': previous_season_id, 'n': int(previous['season_number'])})
    await session.execute(text("DELETE FROM champions_matches WHERE cl_id IN (SELECT id FROM champions_league WHERE season_id=:old OR season=:n)"),
                          {'old': previous_season_id, 'n': int(previous['season_number'])})
    await session.execute(text("DELETE FROM champions_groups WHERE cl_id IN (SELECT id FROM champions_league WHERE season_id=:old OR season=:n)"),
                          {'old': previous_season_id, 'n': int(previous['season_number'])})
    await session.execute(text("DELETE FROM champions_league WHERE season_id=:old OR season=:n"),
                          {'old': previous_season_id, 'n': int(previous['season_number'])})
    # Remove queued events that reference old fixtures before those fixtures are deleted.
    await session.execute(text("DELETE FROM season_events WHERE season_id=:old"), {'old': previous_season_id})
    if old_league_ids:
        await session.execute(text("DELETE FROM matches WHERE season_id=:old OR league_id=ANY(:leagues)"),
                              {'old': previous_season_id, 'leagues': old_league_ids})
    else:
        await session.execute(text("DELETE FROM matches WHERE season_id=:old"), {'old': previous_season_id})
    await session.execute(text("DELETE FROM club_season_finances WHERE league_id=ANY(:leagues)"),
                          {'leagues': old_league_ids or [-1]})
    if old_league_ids:
        await session.execute(text("""
            UPDATE leagues SET season_id=:new,season_number=:n,season_state='PRESEASON',status='ACTIVE',current_round=1,total_rounds=30,finished_at=NULL
            WHERE id=ANY(:league_ids)
        """), {'new': new_id, 'n': next_number, 'league_ids': old_league_ids})
    from app.game.league import round_robin
    from app.game.calendar import get_next_free_date
    from app.game.event_queue import queue_event
    # Use a stable evening kickoff instead of inheriting the arbitrary hour/minute
    # at which an administrator happened to start the season.
    schedule_start = now.replace(hour=20, minute=0, second=0, microsecond=0)
    if schedule_start <= now:
        schedule_start += timedelta(days=1)
    for league in (await session.execute(text("SELECT id FROM leagues WHERE season_id=:s AND name NOT LIKE 'FM27 Quick Test%' ORDER BY id"), {'s': new_id})).mappings().all():
        lid = int(league['id'])
        clubs = [int(x) for x in (await session.execute(text("SELECT club_id FROM league_teams WHERE league_id=:l ORDER BY club_id"), {'l': lid})).scalars().all()]
        if len(clubs) < 16:
            from app.services import create_test_bot_club
            while len(clubs) < 16:
                bot_id, _ = await create_test_bot_club(session)
                await session.execute(text("INSERT INTO league_teams(league_id,club_id) VALUES(:l,:c) ON CONFLICT DO NOTHING"), {'l': lid, 'c': bot_id})
                clubs.append(int(bot_id))
        clubs = clubs[:16]
        await session.execute(text("DELETE FROM matches WHERE league_id=:l AND season_id=:s"), {'l': lid, 's': new_id})
        await session.execute(text("UPDATE league_teams SET played=0,wins=0,draws=0,losses=0,goals_for=0,goals_against=0,points=0 WHERE league_id=:l"), {'l': lid})
        for fixture in round_robin(clubs, True):
            kickoff = schedule_start + timedelta(days=(fixture.round_no - 1) * 2)
            kickoff = await get_next_free_date(session, kickoff, (fixture.home_club_id, fixture.away_club_id), priority=0)
            if not isinstance(kickoff, datetime):
                kickoff = datetime.combine(kickoff, time(20, 0), tzinfo=now.tzinfo)
            match_id = int((await session.execute(text("""
                INSERT INTO matches(season_id,league_id,round,home_club_id,away_club_id,seed,competition_type,scheduled_at)
                VALUES(:s,:l,:r,:h,:a,0,'LEAGUE',CAST(:k AS timestamptz)) RETURNING id
            """), {'s': new_id, 'l': lid, 'r': fixture.round_no, 'h': fixture.home_club_id, 'a': fixture.away_club_id, 'k': kickoff})).scalar_one())
            await session.execute(text("UPDATE matches SET seed=:seed WHERE id=:m"), {'seed': deterministic_match_seed(seed, match_id), 'm': match_id})
            await queue_event(session, 'MATCH_LEAGUE', kickoff, {'season_id': new_id, 'league_id': lid, 'match_id': match_id})
        await session.execute(text("UPDATE leagues SET season_state='ACTIVE',started_at=CAST(:started AS timestamptz) WHERE id=:l"), {'started': now, 'l': lid})
    # Recreate the global competitions for the same persistent season.
    active_league = (await session.execute(text("SELECT id FROM leagues WHERE season_id=:s AND name NOT LIKE 'FM27 Quick Test%' ORDER BY id LIMIT 1"), {'s': new_id})).scalar_one_or_none()
    if active_league is not None:
        from app.game.cup import start_cup_for_league
        cup_id = await start_cup_for_league(session, int(active_league), season_id=new_id)
        await session.execute(text("UPDATE matches SET season_id=:s WHERE competition_type='CUP' AND id IN (SELECT match_id FROM cup_matches WHERE cup_id=:c)"), {'s':new_id,'c':int(cup_id)})
        from app.game.champions_league import start_champions_league
        cl_id = await start_champions_league(session, season_id=next_number)
        await session.execute(text("UPDATE champions_league SET season_id=:s WHERE id=:c"), {'s':new_id,'c':int(cl_id)})
        await session.execute(text("UPDATE matches SET season_id=:s WHERE competition_type='CHAMPIONS' AND id IN (SELECT match_id FROM champions_matches WHERE cl_id=:c)"), {'s':new_id,'c':int(cl_id)})
    await session.execute(text("UPDATE seasons SET state='ACTIVE' WHERE id=:s"), {'s': new_id})
    await session.execute(text("UPDATE season_sequences SET current_season_id=:s WHERE id=:seq"), {'s': new_id, 'seq': sequence_id})
    return new_id


async def handle_season_event(session, event_type, payload):
    if event_type == 'CONTRACT_WINDOW_OPEN':
        await session.execute(text("UPDATE seasons SET state='CONTRACT_WINDOW' WHERE id=:s AND state='OFFSEASON'"), {'s':int(payload['season_id'])})
    elif event_type == 'CONTRACT_WINDOW_CLOSE':
        await advance_contract_window(session, force=True)
    elif event_type == 'SEASON_END':
        await run_season_end(session, int(payload.get('season_id')) if payload.get('season_id') else None)
    elif event_type == 'SEASON_START':
        await advance_contract_window(session, force=True)


async def advance_finished_leagues(session) -> list[int]:
    return [await start_new_season(session)]

export const TELEGRAM_REQUIRED_MESSAGE = 'Откройте приложение из Telegram';

function initData(): string {
  const tg = (window as any).Telegram?.WebApp;
  return typeof tg?.initData === 'string' ? tg.initData.trim() : '';
}

export async function api<T>(path: string, options: RequestInit = {}): Promise<T> {
  const telegramInitData = initData();
  // Do not send unauthenticated API requests when opened in a normal browser.
  if (!telegramInitData) throw new Error(TELEGRAM_REQUIRED_MESSAGE);

  const headers = new Headers(options.headers);
  headers.set('Accept', 'application/json');
  headers.set('X-Telegram-Init-Data', telegramInitData);
  if (options.body && !headers.has('Content-Type')) headers.set('Content-Type', 'application/json');
  let response: Response;
  try { response = await fetch(path, { ...options, headers }); }
  catch { throw new Error('Не удалось связаться с сервером. Проверь интернет и попробуй снова.'); }
  if (!response.ok) {
    let message = `Ошибка API (${response.status})`;
    try { const data = await response.json(); if (data?.detail) message = String(data.detail); } catch { /* non-JSON error */ }
    throw new Error(message);
  }
  return response.json() as Promise<T>;
}

  { id: 'red', label: 'Красный', value: '#ef4444' }, { id: 'blue', label: 'Синий', value: '#3b82f6' },
  { id: 'green', label: 'Зелёный', value: '#22c55e' }, { id: 'yellow', label: 'Жёлтый', value: '#facc15' },
  { id: 'purple', label: 'Фиолетовый', value: '#a855f7' }, { id: 'orange', label: 'Оранжевый', value: '#f97316' },
  { id: 'black', label: 'Чёрный', value: '#171717' }, { id: 'white', label: 'Белый', value: '#f8fafc' },
];
const symbols = [
  { id: 'ball', label: 'Мяч', icon: '⚽' }, { id: 'lion', label: 'Лев', icon: '🦁' },
  { id: 'eagle', label: 'Орёл', icon: '🦅' }, { id: 'wolf', label: 'Волк', icon: '🐺' },
  { id: 'lightning', label: 'Молния', icon: '⚡' }, { id: 'crown', label: 'Корона', icon: '👑' },
  { id: 'star', label: 'Звезда', icon: '⭐' },
];

function ShapeSvg({ shape, color }: { shape: string; color: string }) {
  const common = { fill: color, stroke: 'rgba(255,255,255,.9)', strokeWidth: 2 };
  return <svg viewBox="0 0 64 72" aria-hidden="true" width="44" height="50">
    {shape === 'shield' && <path {...common} d="M8 8 L56 8 L53 39 Q48 56 32 65 Q16 56 11 39 Z" />}
    {shape === 'circle' && <circle {...common} cx="32" cy="36" r="27" />}
    {shape === 'diamond' && <path {...common} d="M32 5 L59 36 L32 67 L5 36 Z" />}
    {shape === 'triangle' && <path {...common} d="M32 5 L60 63 L4 63 Z" />}
    {shape === 'star' && <polygon {...common} points="32,4 40,25 62,26 45,41 51,63 32,50 13,63 19,41 2,26 24,25" />}
    {shape === 'square' && <rect {...common} x="6" y="10" width="52" height="52" rx="5" />}
  </svg>;
}

export default function CreateClubScreen({ onCreated }: { onCreated: () => void }) {
  const [name, setName] = useState('');
  const [shape, setShape] = useState('shield');
  const [color, setColor] = useState('blue');
  const [symbol, setSymbol] = useState('ball');
  const [busy, setBusy] = useState(false);
  const [toast, setToast] = useState('');
  const selectedColor = colors.find(c => c.id === color)?.value ?? '#3b82f6';
  const selectedSymbol = symbols.find(s => s.id === symbol)?.icon ?? '⚽';

  const submit = async (event: React.FormEvent) => {
    event.preventDefault();
    if (name.trim().length < 3 || name.trim().length > 20) {
      setToast('Название клуба должно содержать от 3 до 20 символов.');
      return;
    }
    setBusy(true);
    setToast('');
    try {
      await createClub({ name: name.trim(), emblem_shape: shape, emblem_color: color, emblem_symbol: symbol });
      onCreated();
    } catch (error) {
      setToast(error instanceof Error ? error.message : 'Не удалось создать клуб. Попробуйте ещё раз.');
    } finally {
      setBusy(false);
    }
  };

  return <main className="app-shell create-club-screen">
    <div className="topline"><span>FOOTBALL MANAGER</span><span>НОВАЯ КАРЬЕРА</span></div>
    <header className="create-club-header"><div><small>ТВОЯ ИСТОРИЯ НАЧИНАЕТСЯ ЗДЕСЬ</small><h1>Создай клуб</h1><p className="muted">Придумай имя и собери эмблему своей команды.</p></div></header>
    <form className="card create-club-form" onSubmit={submit}>
      <label htmlFor="club-name">Название клуба</label>
      <input id="club-name" className="club-name-input" value={name} onChange={e => setName(e.target.value.slice(0, 20))} maxLength={20} minLength={3} placeholder="Например, Северные львы" required />
      <small className="muted name-counter">{name.length}/20 · минимум 3 символа</small>

      <h3>1. Форма эмблемы</h3>
      <div className="crest-options">{shapes.map(item => <button type="button" key={item.id} className={`crest-option ${shape === item.id ? 'selected' : ''}`} onClick={() => setShape(item.id)} aria-pressed={shape === item.id}><ShapeSvg shape={item.id} color={selectedColor} /><small>{item.label}</small></button>)}</div>

      <h3>2. Цвет эмблемы</h3>
      <div className="color-options">{colors.map(item => <button type="button" key={item.id} title={item.label} aria-label={item.label} aria-pressed={color === item.id} className={`color-option ${color === item.id ? 'selected' : ''}`} style={{ backgroundColor: item.value }} onClick={() => setColor(item.id)}>{color === item.id ? '✓' : ''}</button>)}</div>
      <p className="muted color-label">{colors.find(c => c.id === color)?.label}</p>

      <h3>3. Символ команды</h3>
      <div className="symbol-options">{symbols.map(item => <button type="button" key={item.id} className={`symbol-option ${symbol === item.id ? 'selected' : ''}`} onClick={() => setSymbol(item.id)} aria-pressed={symbol === item.id}><span>{item.icon}</span><small>{item.label}</small></button>)}</div>

      <div className="crest-preview"><div className="crest-preview-shape"><ShapeSvg shape={shape} color={selectedColor} /><span>{selectedSymbol}</span></div><div><small>ПРЕДПРОСМОТР</small><b>{name.trim() || 'Название клуба'}</b><span className="muted">Твоя будущая эмблема</span></div></div>
      <button className="btn create-club-submit" type="submit" disabled={busy}>{busy ? 'Создаём клуб…' : 'Создать клуб'}</button>
    </form>
    {toast && <div role="status" className="toast" onClick={() => setToast('')}>{toast}</div>}
  </main>;
}
